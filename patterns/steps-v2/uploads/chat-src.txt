class Component extends DCLogic {
  state = { started: false, shown: 0, typing: false, done: false };
  timers = [];

  script() {
    return [
      { side: 'us', text: 'Здравствуйте! Хочу сделать заказ' },
      { side: 'them', text: 'Окей, что вас интересует?' },
      { side: 'us', text: 'Нужна перегородка LOFT' },
      { side: 'them', text: 'Сделаем. Какие сроки?' },
      { side: 'us', text: 'Хотелось бы за неделю' },
      { side: 'them', text: 'Успеем. Смету пришлю сегодня к вечеру' }
    ];
  }

  componentWillUnmount() { this.timers.forEach(clearTimeout); }

  run() {
    const script = this.script();
    let t = 400;
    script.forEach((m, i) => {
      const think = 500 + m.text.length * 22;
      this.timers.push(setTimeout(() => this.setState({ typing: m.side }), t));
      t += think;
      this.timers.push(setTimeout(() => this.setState({ shown: i + 1, typing: false }), t));
      t += 700;
    });
    this.timers.push(setTimeout(() => this.setState({ done: true }), t + 300));
  }

  renderVals() {
    const s = this.state;
    const all = this.script();
    const messages = all.slice(0, s.shown).map((m) => ({
      text: m.text,
      justify: m.side === 'us' ? 'flex-end' : 'flex-start',
      bg: m.side === 'us' ? 'oklch(0.78 0.13 250)' : '#1b1e25',
      fg: m.side === 'us' ? '#0e0f12' : '#e9eaee',
      radius: m.side === 'us' ? '20px 20px 6px 20px' : '20px 20px 20px 6px',
      anim: m.side === 'us' ? 'inRight' : 'inLeft'
    }));
    return {
      messages,
      started: s.started,
      notStarted: !s.started,
      done: s.done,
      typing: !!s.typing,
      typingJustify: s.typing === 'us' ? 'flex-end' : 'flex-start',
      typingBg: s.typing === 'us' ? 'oklch(0.78 0.13 250)' : '#1b1e25',
      typingAnim: s.typing === 'us' ? 'inRight' : 'inLeft',
      typingFg: s.typing === 'us' ? '#0e0f12' : '#e9eaee',
      start: () => {
        this.timers.forEach(clearTimeout);
        this.timers = [];
        this.setState({ started: true, shown: 0, typing: false, done: false }, () => this.run());
      },
      restart: () => {
        this.timers.forEach(clearTimeout);
        this.timers = [];
        this.setState({ started: false, shown: 0, typing: false, done: false });
      }
    };
  }
}
</script>
